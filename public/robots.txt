# ─── All crawlers ────────────────────────────────────────────────────────────
User-agent: *

# Public surfaces — indexable. Exactly mirrors the entries in sitemap.xml.
Allow: /$
Allow: /auth
Allow: /terms
Allow: /privacy
Allow: /faq

# Private, authenticated workspaces — never indexable. These namespaces are
# role-scoped dashboards (student, TPO, recruiter) that require a session
# cookie; crawling them wastes budget, leaks nothing useful, and can surface
# sign-in redirects in search results.
Disallow: /student/
Disallow: /tpo/
Disallow: /company/
Disallow: /messages
Disallow: /dashboard

# API / data endpoints — never crawled.
Disallow: /api/

# Build and asset internals that should not appear in results.
Disallow: /assets/

# ─── Explicit allow for crawlers that read social/SEO preview assets ─────────
User-agent: Twitterbot
Allow: /

User-agent: facebookexternalhit
Allow: /
